// Fatihhan Tekstil site üreticisi. build(spec, media) -> [{path, content}]
// spec: satır başına "slug|Ad|SERİ=kodlar;SERİ=kodlar". media: dosya adı -> {url, w, h}
function build(spec, media) {
  var BASE = 'https://www.fatihhantekstil.com';
  var TEL = '0534 565 89 99';
  var WA = 'https://wa.me/905345658999?text=';
  var esc = function (s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/"/g, '&quot;'); };
  var slug = function (s) {
    var m = { 'İ': 'i', 'I': 'i', 'Ş': 's', 'Ğ': 'g', 'Ü': 'u', 'Ö': 'o', 'Ç': 'c', 'ı': 'i', 'ş': 's', 'ğ': 'g', 'ü': 'u', 'ö': 'o', 'ç': 'c' };
    return s.replace(/[İIŞĞÜÖÇışğüöç]/g, function (c) { return m[c]; }).toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  };
  var wa = function (t) { return WA + encodeURIComponent(t); };
  var pic = function (name, w, h) {
    var f = media[name];
    if (!f) return '';
    if (f.url.indexOf('static.wixstatic.com') < 0) return f.url;
    var r = Math.min(1, w / f.w);
    var ww = Math.round(f.w * r), hh = Math.round(f.h * r);
    return f.url + '/v1/fill/w_' + ww + ',h_' + hh + ',al_c,q_85,enc_auto/' + name;
  };

  // ---- veriyi aç ----

    var p = line.split('|');
    var col = { slug: p[0], name: p[1], series: [], n: 0, cover: p[0] + '-kumas-koleksiyonu.webp' };

      var i = part.indexOf('=');
      var sname = part.slice(0, i);
      var join = /-$/.test(sname) ? '' : ' ';
      var items = [];
      part.slice(i + 1).split(',').forEach(function (tok) {
        var r = tok.match(/^(\d+)\.\.(\d+)$/);
        var list = [tok];
        if (r) {
          list = [];
          for (var n = parseInt(r[1], 10); n <= parseInt(r[2], 10); n++) {
            var t = String(n);
            while (t.length < r[1].length) t = '0' + t;
            list.push(t);
          }
        }
        list.forEach(function (t) {
          var code = sname + join + t;
          var s = slug(code);
          seen[s] = (seen[s] || 0) + 1;
          var id = seen[s] > 1 ? s + '-' + seen[s] : s;
          items.push({ code: code, id: id, file: id + '-dosemelik-kumas.webp', url: 'kumas-' + id + '.html' });
        });
      });
      col.series.push({ name: sname.replace(/ V-$/, ''), items: items });
      col.n += items.length;
    });
    col.url = 'kartela-' + col.slug + '.html';
    return col;
  });

  cols.forEach(function (c) { bySlug[c.slug] = c; });

      '<link rel="canonical" href="' + BASE + '/' + o.path + '">\n<meta name="robots" content="index,follow,max-image-preview:large">\n<link rel="icon" href="/favicon.svg" type="image/svg+xml">\n' +
